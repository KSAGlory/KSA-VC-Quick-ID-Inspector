{$CLEO .cs}
{$NOSOURCE}

script_name "KSA_VCI"

int enabled = true
int playerChar
int target = -1
int modelId
int health
int vehicle
int vehicleModel
int vehicleHealth
int toggleNotice = 0
int pointer
int vehicleKey
int vehicleName
float playerX
float playerY
float playerZ
int nextScan = 0

vehicleKey = get_label_pointer @VehicleKeyBuffer
vehicleName = get_label_pointer @VehicleNameBuffer
TimerA = 0
TimerB = 0

while true
    wait 0

    if
        not is_player_playing 0
    then
        TimerA = 0
        target = -1
        continue
    end

    playerChar = get_player_char 0

    if
        test_cheat "KSAID"
    then
        if
            enabled == true
        then
            enabled = false
            toggleNotice = 2
        else
            enabled = true
            toggleNotice = 1
        end
        TimerB = 0
        target = -1
    end

    use_text_commands true

    if and
        toggleNotice <> 0
        TimerB < 1800
    then
        gosub @DrawNotice
    else
        toggleNotice = 0
    end

    if
        enabled == false
    then
        continue
    end

    if
        TimerA < 4000
    then
        gosub @DrawIntro
        continue
    end

    if
        is_char_sitting_in_any_car playerChar
    then
        target = -1
        vehicle = store_car_char_is_in_no_save playerChar
        vehicleModel = get_car_model vehicle
        vehicleHealth = get_car_health vehicle
        // VC's vehicle-name opcode indexes the original vehicle model table.
        if and
            vehicleModel >= 130
            vehicleModel <= 236
        then
            vehicleKey = get_name_of_vehicle_model vehicleModel
            vehicleName = get_text_label_string vehicleKey
        end
        gosub @DrawVehicle
        continue
    end

    if
        not is_button_pressed 0 Button.RightShoulder1
    then
        target = -1
        nextScan = 0
        continue
    end

    pointer = get_game_timer
    if
        pointer >= nextScan
    then
        nextScan = pointer
        nextScan += 100
        target = -1
        playerX, playerY, playerZ = get_char_coordinates playerChar
        // Restart CLEO's ped enumeration, then check VC's actual aim target.
        if
            target = get_random_char_in_sphere_no_save_recursive playerX playerY playerZ 100.0 false true
        then
            while true
                if
                    is_player_targetting_char 0 target
                then
                    break
                end
                if
                    not target = get_random_char_in_sphere_no_save_recursive playerX playerY playerZ 100.0 true true
                then
                    target = -1
                    break
                end
            end
        end
    end

    if
        target >= 0
    then
        if
            does_char_exist target
        then
            if
                is_player_targetting_char 0 target
            then
                pointer = get_ped_pointer target
                if
                    pointer <> 0
                then
                    // plugin-sdk VC CEntity: short m_nModelIndex at offset 0x5C.
                    pointer += 0x5C
                    modelId = read_memory pointer 2 false
                    health = get_char_health target
                    gosub @DrawPed
                end
            end
        else
            target = -1
        end
    end
end
